cmake_minimum_required(VERSION 3.25)
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(picoworkbench_project_gate C CXX ASM)
pico_sdk_init()
configure_file(generated.h.in generated.h @ONLY)
add_library(project_values STATIC right/value.c)
target_include_directories(project_values PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(project_values pico_stdlib)
foreach(name project_a project_b)
    add_executable(${name} main.c left/value.c "${CMAKE_CURRENT_BINARY_DIR}/generated.h")
    target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(${name} pico_stdlib project_values hardware_pio)
    pico_enable_stdio_uart(${name} 1)
    pico_enable_stdio_usb(${name} 0)
endforeach()
target_compile_definitions(project_b PRIVATE PROJECT_B=1)
pico_generate_pio_header(project_b "${CMAKE_CURRENT_SOURCE_DIR}/gate.pio")
