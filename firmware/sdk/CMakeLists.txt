cmake_minimum_required(VERSION 3.25)
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(rp2040_sim_sdk_fixtures C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.2.0")
    message(FATAL_ERROR "This fixture is validated with Pico SDK 2.2.0")
endif()
foreach(name sdk_blink sdk_uart sdk_flash sdk_dashboard)
    add_executable(${name} ${name}.c)
    target_link_libraries(${name} pico_stdlib)
    target_link_options(${name} PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${name}.map")
    pico_enable_stdio_usb(${name} 0)
    pico_enable_stdio_uart(${name} 0)
    pico_set_float_implementation(${name} compiler)
    pico_set_double_implementation(${name} compiler)
    target_compile_definitions(${name} PRIVATE PICO_CXX_DISABLE_ALLOCATION_OVERRIDES=1)
    set_target_properties(${name} PROPERTIES LINKER_LANGUAGE C)
endforeach()
pico_enable_stdio_uart(sdk_uart 1)
pico_enable_stdio_uart(sdk_flash 1)
target_link_libraries(sdk_flash hardware_flash)
pico_enable_stdio_uart(sdk_dashboard 1)
target_link_libraries(sdk_dashboard hardware_i2c hardware_spi hardware_adc)
