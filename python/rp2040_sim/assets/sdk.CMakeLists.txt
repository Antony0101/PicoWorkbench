cmake_minimum_required(VERSION 3.25)
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(rp2040_sim_sdk_editor C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.2.0")
    message(FATAL_ERROR "The supported simulator profile requires Pico SDK 2.2.0")
endif()
add_executable(editor editor.c)
target_link_libraries(editor pico_stdlib hardware_flash)
pico_enable_stdio_usb(editor 0)
pico_enable_stdio_uart(editor 1)
pico_set_float_implementation(editor compiler)
pico_set_double_implementation(editor compiler)
target_compile_definitions(editor PRIVATE PICO_CXX_DISABLE_ALLOCATION_OVERRIDES=1)
set_target_properties(editor PROPERTIES LINKER_LANGUAGE C)
